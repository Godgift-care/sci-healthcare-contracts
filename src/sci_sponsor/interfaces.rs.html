1//! The slices of the other contracts this contract depends on.
2//!
3//! Declared locally, over primitives only, so the contracts share no types
4//! and can be upgraded independently.
5
6use soroban_sdk::{contractclient, contracterror, Address, BytesN, Env, String};
7
8/// The slice of `sci-registry` this contract depends on.
9#[allow(dead_code)]
10#[contractclient(name = "RegistryClient")]
11pub trait RegistryInterface {
12    /// The country a clinic is registered in, as an ISO 3166-1 alpha-2
13    /// code. Empty when the clinic is not registered.
14    fn get_provider_country(env: Env, provider_addr: Address) -> String;
15}
16
17/// The slice of `sci-voucher` this contract depends on.
18#[allow(dead_code)]
19#[contractclient(name = "VoucherClient")]
20pub trait VoucherInterface {
21    /// The fields of a voucher this contract needs to match it, in one
22    /// call.
23    ///
24    /// `(beneficiary_ref, funder, provider, service_code, amount, status,
25    /// created_at, expires_at)`, where `status` is the voucher's
26    /// `VoucherStatus` as a number: 0 `Funded`, 1 `Claimed`, 2 `Attested`,
27    /// 3 `Settled`, 4 `Disputed`, 5 `Refunded`. `None` when the voucher
28    /// does not exist.
29    #[allow(clippy::type_complexity)]
30    fn voucher_summary(
31        env: Env,
32        voucher_id: u64,
33    ) -> Option<(BytesN<32>, Address, Address, u32, i128, u32, u64, u64)>;
34
35    /// Escrows `amount` for one service at one active provider and returns
36    /// the new voucher's id.
37    #[allow(clippy::too_many_arguments)]
38    fn create_voucher(
39        env: Env,
40        funder: Address,
41        beneficiary_ref: BytesN<32>,
42        provider: Address,
43        service_code: u32,
44        amount: i128,
45        expires_at: u64,
46    ) -> Result<u64, VoucherError>;
47}
48
49/// The voucher contract's error codes, mirrored so this contract can read
50/// the result of a cross-contract call without sharing types with it.
51///
52/// The voucher contract is a separate deployment with its own numbering.
53/// Only the codes a sponsorship can run into are listed; anything else comes
54/// back as `VoucherCallFailed`.
55#[contracterror]
56#[derive(Clone, Copy, Debug, PartialEq, Eq)]
57#[repr(u32)]
58pub enum VoucherError {
59    /// `ProviderNotActive` in the voucher contract.
60    ProviderNotActive = 5,
61    /// `ServiceNotOffered` in the voucher contract.
62    ServiceNotOffered = 6,
63    /// `AmountBelowPrice` in the voucher contract.
64    AmountBelowPrice = 7,
65    /// `ServiceUnavailable` in the voucher contract.
66    ServiceUnavailable = 18,
67}
