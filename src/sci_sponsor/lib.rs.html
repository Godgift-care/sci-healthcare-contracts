8//! # SCI Healthcare — Sponsorship desk
9//!
10//! Lets a sponsor put a budget behind someone else's voucher. The sponsor
11//! deposits a budget, sets the rules it wants, and the desk adds a *second*
12//! voucher for the same care — funded by the sponsor — whenever a voucher
13//! qualifies.
14//!
15//! ## Why a second voucher and not a bigger one
16//!
17//! The match is an ordinary voucher, created through the voucher contract
18//! and escrowed there. Nothing about the care lifecycle is re-implemented
19//! here: the clinic claims it, an attester confirms it, the funder can
20//! dispute it, and the money is settled or refunded by the same code that
21//! already handles every other voucher. This contract only decides *whether*
22//! a match happens and *how much* it costs the sponsor.
23//!
24//! ## Why the money sits here
25//!
26//! The matching voucher is funded by this contract, so the sponsor's budget
27//! is escrowed by the voucher contract exactly like a funder's own money.
28//! The sponsor's budget is the difference between what it deposited and
29//! what it has committed to matches; whatever is left can be withdrawn.
30//!
31//! ## What this contract is not
32//!
33//! It is not a source of care. A match never creates a claimable right on
34//! its own: the money is only ever spent when the clinic delivers and an
35//! attester confirms it, exactly as for any other voucher. The abuse cases
36//! this design has to answer are written up in
37//! the `sponsor` section of `docs/developers/contracts.md`.

69/// Basis points in one hundred per cent.
70pub const BPS_DENOMINATOR: i128 = 10_000;
71/// Ceiling on the match ratio: twenty times the matched amount.
72pub const MAX_MATCH_BPS: u32 = 200_000;
73/// Most matches one sponsor may have open at once.
74///
75/// An open match is one whose matching voucher has neither settled nor been
76/// refunded. The desk checks each of them when the sponsor's books are next
77/// touched, so the list is kept short enough for that to stay cheap.
78pub const MAX_OPEN_MATCHES: u32 = 20;
79/// The voucher's `VoucherStatus` as a number.
80const VOUCHER_STATUS_FUNDED: u32 = 0;
81const VOUCHER_STATUS_SETTLED: u32 = 3;
82const VOUCHER_STATUS_REFUNDED: u32 = 5;

120    /// Adds to a sponsor's budget.
121    ///
122    /// The money moves from the sponsor's own account under the sponsor's own
123    /// signature, into this contract, which holds it on the sponsor's
124    /// behalf until it is matched or withdrawn.
125    pub fn deposit(env: Env, sponsor: Address, amount: i128) -> Result<(), SponsorError> {
126        let config = Self::read_config(&env)?;
127        if config.paused {
128            return Err(SponsorError::Paused);
129        }
130        sponsor.require_auth();
131        if amount <= 0 {
132            return Err(SponsorError::InvalidAmount);
133        }
134
135        // The money moves first: if the transfer fails, nothing is recorded.
136        let desk = env.current_contract_address();
137        token::Client::new(&env, &config.token).transfer(&sponsor, &desk, &amount);
138
139        let mut account = read_account(&env, &sponsor);
140        let mut totals = read_totals(&env);
141        record_movement(&mut account, &mut totals, amount, 0, 0, 0)?;
142        write_account(&env, &account);
143        write_totals(&env, &totals);
144
145        BudgetDeposited {
146            sponsor,
147            amount,
148            available: available_of(&account)?,
149        }
150        .publish(&env);
151        Ok(())
152    }

178    /// Matches a voucher: the sponsor's budget pays for a second voucher
179    /// for the same care.
180    ///
181    /// The matching voucher is created by the voucher contract and funded by
182    /// this contract, so it is escrowed, claimed, attested and settled like
183    /// any other. It carries the same beneficiary, the same clinic and the
184    /// same service as the voucher that was matched, and it expires with it.
185    ///
186    /// The two are linked both ways: `get_match` takes the voucher that was
187    /// matched, `get_match_of` takes the matching voucher.
188    ///
189    /// Errors:
190    ///
191    /// - `RulesNotSet` — the sponsor has not set any rules.
192    /// - `VoucherNotFound` — there is no such voucher.
193    /// - `VoucherNotFunded` — the voucher is not `Funded`.
194    /// - `VoucherExpired` — the voucher has expired.
195    /// - `AlreadyMatched` — the voucher has already been matched.
196    /// - `SelfMatchNotAllowed` — the sponsor funded the voucher itself, or
197    ///   this contract did.
198    /// - `CountryNotEligible`, `ServiceNotEligible` — the clinic or the
199    ///   service is outside the sponsor's rules.
200    /// - `VoucherTooNew`, `VoucherTooSmall` — the voucher is outside the
201    ///   sponsor's rules.
202    /// - `MatchBelowPrice` — the match is worth less than the service, and
203    ///   a voucher cannot be funded below the price of the care it pays for.
204    /// - `BudgetExhausted` — the sponsor's budget cannot cover the match.
205    pub fn match_voucher(env: Env, sponsor: Address, voucher_id: u64) -> Result<u64, SponsorError> {
206        let config = Self::read_config(&env)?;
207        if config.paused {
208            return Err(SponsorError::Paused);
209        }
210        sponsor.require_auth();
211
212        let mut account = read_account(&env, &sponsor);
213        let mut totals = read_totals(&env);
214
215        let rules = account.rules.clone();
216        if rules.match_bps == 0 {
217            return Err(SponsorError::RulesNotSet);
218        }
219
220        if read_match(&env, voucher_id).is_some() {
221            return Err(SponsorError::AlreadyMatched);
222        }
223
224        let vouchers = VoucherClient::new(&env, &config.voucher);
225        let (
226            beneficiary_ref,
227            funder,
228            provider,
229            service_code,
230            amount,
231            status,
232            created_at,
233            expires_at,
234        ) = vouchers
235            .voucher_summary(&voucher_id)
236            .ok_or(SponsorError::VoucherNotFound)?;
237
238        if status != VOUCHER_STATUS_FUNDED {
239            return Err(SponsorError::VoucherNotFunded);
240        }
241        let now = env.ledger().timestamp();
242        if expires_at <= now {
243            return Err(SponsorError::VoucherExpired);
244        }
245
246        // The sponsor may not match a voucher it funded itself, and may not
247        // match one this contract funded: either way the match would pay the
248        // payer of the match.
249        let desk = env.current_contract_address();
250        if funder == sponsor || funder == desk {
251            return Err(SponsorError::SelfMatchNotAllowed);
252        }
253
254        if now.saturating_sub(created_at) < rules.min_voucher_age_secs {
255            return Err(SponsorError::VoucherTooNew);
256        }
257        if amount < rules.min_voucher_amount {
258            return Err(SponsorError::VoucherTooSmall);
259        }
260
261        // The clinic has to be in one of the sponsor's countries, and the
262        // service has to be one the sponsor matches. What the clinic is
263        // allowed to bill for, and what the service costs, is the voucher
264        // contract's business: it checks the price and the clinic's status
265        // when it creates the matching voucher, and its answer comes back
266        // as a typed error.
267        if !rules.eligible_countries.is_empty() {
268            let country =
269                RegistryClient::new(&env, &config.registry).get_provider_country(&provider);
270            if !rules.eligible_countries.contains(&country) {
271                return Err(SponsorError::CountryNotEligible);
272            }
273        }
274        if !rules.eligible_service_codes.is_empty()
275            && !rules.eligible_service_codes.contains(service_code)
276        {
277            return Err(SponsorError::ServiceNotEligible);
278        }
279
280        let match_amount = Self::match_amount(amount, &rules)?;
281        if match_amount <= 0 {
282            return Err(SponsorError::MatchTooSmall);
283        }
284
285        let available = Self::budget_for(&env, &config, &mut account, &mut totals, match_amount)?;
286        if available < match_amount {
287            return Err(SponsorError::BudgetExhausted);
288        }
289        if account.open_matches.len() >= MAX_OPEN_MATCHES {
290            return Err(SponsorError::TooManyOpenMatches);
291        }
292
293        // The matching voucher, funded by this contract and escrowed by the
294        // voucher contract.
295        let matched_id = Self::create_matching_voucher(
296            &env,
297            &config,
298            &beneficiary_ref,
299            &provider,
300            service_code,
301            match_amount,
302            expires_at,
303        )?;
304
305        let record = MatchRecord {
306            original_voucher_id: voucher_id,
307            matched_voucher_id: matched_id,
308            sponsor: sponsor.clone(),
309            amount: match_amount,
310            provider: provider.clone(),
311            service_code,
312            matched_at: now,
313        };
314        write_match(&env, &record);
315        write_original_of(&env, matched_id, voucher_id);
316
317        account.open_matches.push_back(OpenMatch {
318            matched_voucher_id: matched_id,
319            amount: match_amount,
320        });
321        record_movement(&mut account, &mut totals, 0, match_amount, 0, 0)?;
322        write_account(&env, &account);
323        write_totals(&env, &totals);
324
325        MatchCreated {
326            sponsor,
327            original_voucher_id: voucher_id,
328            matched_voucher_id: matched_id,
329            provider,
330            service_code,
331            amount: match_amount,
332            matched_at: now,
333        }
334        .publish(&env);
335        Ok(matched_id)
336    }

338    /// Gives the sponsor back money it has not matched.
339    ///
340    /// Only the unused budget can be withdrawn: money that is escrowed in a
341    /// matching voucher is not the sponsor's to take back while the care
342    /// could still be delivered.
343    pub fn withdraw(env: Env, sponsor: Address, amount: i128) -> Result<(), SponsorError> {
344        let config = Self::read_config(&env)?;
345        sponsor.require_auth();
346        if amount <= 0 {
347            return Err(SponsorError::InvalidAmount);
348        }
349
350        let mut account = read_account(&env, &sponsor);
351        let mut totals = read_totals(&env);
352        // A refunded match is money that came back: it is unused budget
353        // again. The books are brought up to date when the budget looks too
354        // small for this withdrawal, which is the only time it can matter.
355        let mut available = available_of(&account)?;
356        if amount > available {
357            Self::reconcile(&env, &config, &mut account, &mut totals)?;
358            available = available_of(&account)?;
359        }
360        if amount > available {
361            return Err(SponsorError::WithdrawExceedsBudget);
362        }
363
364        let desk = env.current_contract_address();
365        token::Client::new(&env, &config.token).transfer(&desk, &sponsor, &amount);
366
367        record_movement(&mut account, &mut totals, 0, 0, 0, amount)?;
368        write_account(&env, &account);
369        write_totals(&env, &totals);
370
371        BudgetWithdrawn {
372            sponsor,
373            amount,
374            available: available_of(&account)?,
375        }
376        .publish(&env);
377        Ok(())
378    }

380    /// Brings a sponsor's books up to date with the matching vouchers that
381    /// have since settled or been refunded.
382    ///
383    /// A settled match is money that became care: it stays spent. A refunded
384    /// match is money that came back: it is available again. The desk cannot
385    /// see a refund happen, so it checks the matching vouchers themselves.
386    ///
387    /// Permissionless: anyone may call it, and `match_voucher` and
388    /// `withdraw` call it for the sponsor they are acting for.
389    pub fn reconcile_sponsor(env: Env, sponsor: Address) -> Result<(), SponsorError> {
390        let config = Self::read_config(&env)?;
391        let mut account = read_account(&env, &sponsor);
392        let mut totals = read_totals(&env);
393        Self::reconcile(&env, &config, &mut account, &mut totals)?;
394        write_account(&env, &account);
395        write_totals(&env, &totals);
396        Ok(())
397    }

483    /// Read-only solvency check: `(available_budget, token_balance)`.
484    ///
485    /// `available_budget` is every sponsor's unused budget added together.
486    /// `token_balance` is this contract's actual balance of the settlement
487    /// token. The two are equal when the accounting is right; a balance that
488    /// is larger means a matching voucher was refunded and the books have
489    /// not been reconciled yet.
490    ///
491    /// No authorisation: anyone may run the check.
492    pub fn budget_summary(env: Env) -> Result<(i128, i128), SponsorError> {
493        let config = Self::read_config(&env)?;
494        let totals = read_totals(&env);
495        let available = totals
496            .deposited
497            .checked_sub(totals.committed)
498            .and_then(|v| v.checked_add(totals.refunded))
499            .and_then(|v| v.checked_sub(totals.withdrawn))
500            .ok_or(SponsorError::MathOverflow)?;
501        let balance =
502            token::Client::new(&env, &config.token).balance(&env.current_contract_address());
503        Ok((available, balance))
504    }

533    /// The rules the desk will accept.
534    fn check_rules(rules: &Rules) -> Result<(), SponsorError> {
535        if rules.match_bps == 0 || rules.match_bps > MAX_MATCH_BPS {
536            return Err(SponsorError::InvalidRules);
537        }
538        if rules.cap_per_voucher <= 0 {
539            return Err(SponsorError::InvalidRules);
540        }
541        if rules.min_voucher_amount < 0 {
542            return Err(SponsorError::InvalidRules);
543        }
544        // ISO 3166-1 alpha-2 is always two characters.
545        for country in rules.eligible_countries.iter() {
546            if country.len() != 2 {
547                return Err(SponsorError::InvalidRules);
548            }
549        }
550        Ok(())
551    }

553    /// What the sponsor adds for a voucher: their share of the amount, capped.
554    fn match_amount(amount: i128, rules: &Rules) -> Result<i128, SponsorError> {
555        let share = amount
556            .checked_mul(rules.match_bps as i128)
557            .ok_or(SponsorError::MathOverflow)?
558            .checked_div(BPS_DENOMINATOR)
559            .ok_or(SponsorError::MathOverflow)?;
560        Ok(core::cmp::min(share, rules.cap_per_voucher))
561    }

563    /// The sponsor's unused budget, bringing the books up to date first when
564    /// that can change the answer.
565    ///
566    /// A matching voucher that was refunded gives the money back, but the
567    /// desk cannot see a refund happen: it has to ask the voucher contract
568    /// about every open match, one call each. That is only worth paying for
569    /// when the budget looks too small for what is about to happen, or when
570    /// the desk needs room for another open match. A sponsor with budget to
571    /// spare pays nothing for it.
572    fn budget_for(
573        env: &Env,
574        config: &Config,
575        account: &mut SponsorAccount,
576        totals: &mut Totals,
577        needed: i128,
578    ) -> Result<i128, SponsorError> {
579        let available = available_of(account)?;
580        if available >= needed && account.open_matches.len() < MAX_OPEN_MATCHES {
581            return Ok(available);
582        }
583        Self::reconcile(env, config, account, totals)?;
584        available_of(account)
585    }

587    /// Creates the matching voucher, funded by this contract.
588    fn create_matching_voucher(
589        env: &Env,
590        config: &Config,
591        beneficiary_ref: &BytesN<32>,
592        provider: &Address,
593        service_code: u32,
594        amount: i128,
595        expires_at: u64,
596    ) -> Result<u64, SponsorError> {
597        let desk = env.current_contract_address();
598        // The voucher contract moves the money out of this contract's
599        // balance, so the token call it makes on the way has to be
600        // authorised by this contract. The authorisation names that one
601        // call and no other: this transfer, of this amount, from this
602        // contract to the voucher's escrow.
603        Self::authorize_token_transfer(env, config, &desk, amount);
604
605        match VoucherClient::new(env, &config.voucher).try_create_voucher(
606            &desk,
607            beneficiary_ref,
608            provider,
609            &service_code,
610            &amount,
611            &expires_at,
612        ) {
613            Ok(Ok(id)) => Ok(id),
614            Ok(Err(_)) => Err(SponsorError::VoucherCallFailed),
615            Err(Ok(VoucherError::AmountBelowPrice)) => Err(SponsorError::MatchBelowPrice),
616            Err(Ok(VoucherError::ProviderNotActive)) => Err(SponsorError::ProviderNotEligible),
617            Err(Ok(VoucherError::ServiceNotOffered)) => Err(SponsorError::ServiceNotEligible),
618            Err(Ok(VoucherError::ServiceUnavailable)) => Err(SponsorError::ServiceUnavailable),
619            Err(Err(_)) => Err(SponsorError::VoucherCallFailed),
620        }
621    }

623    /// Authorises the token transfer the voucher contract makes on this
624    /// contract's behalf.
625    ///
626    /// The voucher contract escrows the money by moving it from the funder's
627    /// account to its own. When the funder is this contract, that transfer
628    /// is made by the voucher contract, not by this one, so this contract
629    /// has to authorise it explicitly. The authorisation covers exactly one
630    /// call: `transfer` of `amount` from this contract to the voucher
631    /// contract, on the configured token.
632    fn authorize_token_transfer(env: &Env, config: &Config, from: &Address, amount: i128) {
633        // The voucher's escrow is the voucher contract's own address, and
634        // the token client takes the destination as a muxed address.
635        let escrow = MuxedAddress::from(&config.voucher);
636        env.authorize_as_current_contract(vec![
637            env,
638            InvokerContractAuthEntry::Contract(SubContractInvocation {
639                context: ContractContext {
640                    contract: config.token.clone(),
641                    fn_name: Symbol::new(env, "transfer"),
642                    args: vec![
643                        env,
644                        from.clone().into_val(env),
645                        escrow.into_val(env),
646                        amount.into_val(env),
647                    ],
648                },
649                sub_invocations: vec![env],
650            }),
651        ]);
652    }

654    /// Brings a sponsor's books up to date with the matching vouchers that
655    /// have since settled or been refunded.
656    fn reconcile(
657        env: &Env,
658        config: &Config,
659        account: &mut SponsorAccount,
660        totals: &mut Totals,
661    ) -> Result<(), SponsorError> {
662        if account.open_matches.is_empty() {
663            return Ok(());
664        }
665        let vouchers = VoucherClient::new(env, &config.voucher);
666        let mut still_open = Vec::new(env);
667        let mut refunded = 0i128;
668
669        for open in account.open_matches.iter() {
670            let Some((_, _, _, _, _, status, _, _)) =
671                vouchers.voucher_summary(&open.matched_voucher_id)
672            else {
673                return Err(SponsorError::VoucherNotFound);
674            };
675            match status {
676                VOUCHER_STATUS_SETTLED => {
677                    // The money became care: it stays spent.
678                    MatchSettled {
679                        sponsor: account.sponsor.clone(),
680                        matched_voucher_id: open.matched_voucher_id,
681                        amount: open.amount,
682                    }
683                    .publish(env);
684                }
685                VOUCHER_STATUS_REFUNDED => {
686                    // The money came back: it is available again.
687                    refunded = refunded
688                        .checked_add(open.amount)
689                        .ok_or(SponsorError::MathOverflow)?;
690                    MatchRefunded {
691                        sponsor: account.sponsor.clone(),
692                        matched_voucher_id: open.matched_voucher_id,
693                        amount: open.amount,
694                    }
695                    .publish(env);
696                }
697                _ => still_open.push_back(open.clone()),
698            }
699        }
700
701        account.open_matches = still_open;
702        if refunded > 0 {
703            record_movement(account, totals, 0, 0, refunded, 0)?;
704        }
705        Ok(())
706    }
