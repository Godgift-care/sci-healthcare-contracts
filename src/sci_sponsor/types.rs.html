3/// Addresses and switches this contract works with.
4#[contracttype]
5#[derive(Clone, Debug, PartialEq, Eq)]
6pub struct Config {
7    /// Address allowed to pause the desk and to hand the role on.
8    pub admin: Address,
9    /// The provider registry: service prices, service availability and the
10    /// country a clinic is registered in.
11    pub registry: Address,
12    /// The voucher contract that escrows the money and runs the care
13    /// lifecycle. A matching voucher is an ordinary voucher, funded by this
14    /// contract.
15    pub voucher: Address,
16    /// The settlement token both contracts move.
17    pub token: Address,
18    /// While true, deposits and matches are rejected for everyone. The
19    /// admin can still withdraw: pausing stops money going out, it does not
20    /// trap money already here.
21    pub paused: bool,
22}

24/// The conditions a sponsor sets for the vouchers it will match.
25///
26/// A sponsor picks the rules it wants and the desk honours them on every
27/// match. The rules are the sponsor's own: one sponsor's cap never applies
28/// to another's money.
29#[contracttype]
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub struct Rules {
32    /// The match as a share of the matched voucher's amount, in basis
33    /// points. `10_000` is one-for-one: the sponsor adds as much as the
34    /// funder put in. `20_000` is two-for-one.
35    pub match_bps: u32,
36    /// The most the sponsor will add for any one voucher, whatever the
37    /// ratio works out to.
38    pub cap_per_voucher: i128,
39    /// The least a voucher may be worth to qualify. Zero means no minimum.
40    pub min_voucher_amount: i128,
41    /// How long a voucher must have been funded before it qualifies, in
42    /// seconds. Zero means a voucher can be matched as soon as it exists.
43    ///
44    /// This is the anti-farming wait: a voucher that appeared a second ago
45    /// and is about to be claimed by the clinic that funded it cannot be
46    /// matched in the same breath.
47    pub min_voucher_age_secs: u64,
48    /// The countries a matched clinic may be registered in, as ISO 3166-1
49    /// alpha-2 codes. Empty means any country qualifies.
50    pub eligible_countries: Vec<String>,
51    /// The service codes that qualify. Empty means any service qualifies.
52    pub eligible_service_codes: Vec<u32>,
53}

55/// A match the sponsor has made whose matching voucher is still open.

68/// One sponsor's budget and rules.
69#[contracttype]
70#[derive(Clone, Debug, PartialEq, Eq)]
71pub struct SponsorAccount {
72    /// The sponsor this account belongs to.
73    pub sponsor: Address,
74    /// Everything the sponsor has ever deposited.
75    pub deposited: i128,
76    /// Everything the sponsor has ever committed to a match. A refunded
77    /// match stays counted here; the refund is counted in `refunded`.
78    pub committed: i128,
79    /// Money that came back from a matching voucher and is available again.
80    pub refunded: i128,
81    /// Everything the sponsor has ever withdrawn.
82    pub withdrawn: i128,
83    /// The rules the sponsor set. A `match_bps` of zero means no rules have
84    /// been set yet: a ratio of zero is not a rule the desk accepts, so it
85    /// can stand for "not set" without a second field.
86    pub rules: Rules,
87    /// Matches whose matching voucher is still open.
88    pub open_matches: Vec<OpenMatch>,
89}

142#[contracterror]
143#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
144#[repr(u32)]
145pub enum SponsorError {
146    AlreadyInitialized = 1,
147    NotInitialized = 2,
148    NotAuthorized = 3,
149    /// The amount deposited or withdrawn was not positive.
150    InvalidAmount = 4,
151    /// The rules are outside what the desk will accept.
152    InvalidRules = 5,
153    /// The sponsor has not set any rules yet, so nothing is eligible.
154    RulesNotSet = 6,
155    /// The voucher does not exist in the voucher contract.
156    VoucherNotFound = 7,
157    /// The voucher is not `Funded`, so it cannot be matched.
158    VoucherNotFunded = 8,
159    /// The voucher has expired.
160    VoucherExpired = 9,
161    /// The voucher has already been matched.
162    AlreadyMatched = 10,
163    /// The sponsor tried to match a voucher it funded itself, or one this
164    /// contract funded.
165    SelfMatchNotAllowed = 11,
166    /// The clinic is not registered in one of the sponsor's countries.
167    CountryNotEligible = 12,
168    /// The service is not one the sponsor matches.
169    ServiceNotEligible = 13,
170    /// The voucher was funded too recently to qualify.
171    VoucherTooNew = 14,
172    /// The voucher is worth less than the sponsor's minimum.
173    VoucherTooSmall = 15,
174    /// The match works out below the service price, and a voucher cannot be
175    /// funded below the price of the care it pays for.
176    MatchBelowPrice = 16,
177    /// The sponsor's budget cannot cover this match.
178    BudgetExhausted = 17,
179    /// The withdrawal is larger than the sponsor's unused budget.
180    WithdrawExceedsBudget = 18,
181    /// The sponsor already has as many open matches as the desk allows.
182    TooManyOpenMatches = 19,
183    /// The desk is paused; deposits and matches are rejected.
184    Paused = 20,
185    /// The voucher contract refused the matching voucher.
186    VoucherCallFailed = 21,
187    /// The clinic is not active in the registry, so it cannot be paid.
188    ProviderNotEligible = 22,
189    /// The clinic has paused this service, so it cannot be funded.
190    ServiceUnavailable = 23,
191    /// The match works out to nothing, because the ratio is small and the
192    /// voucher is small.
193    MatchTooSmall = 24,
194    /// A counter would have to exceed the largest amount the token can
195    /// represent.
196    MathOverflow = 25,
197    /// There is no admin nomination waiting to be accepted.
198    NoPendingAdmin = 26,
199}

201/// Maps the shared helpers' failures onto this contract's error numbers, so
202/// callers keep seeing the same codes they did before the helpers were shared.
203impl From<sci_common::CommonError> for SponsorError {
204    fn from(error: sci_common::CommonError) -> Self {
205        match error {
206            sci_common::CommonError::NotAuthorized => SponsorError::NotAuthorized,
207            sci_common::CommonError::NoPendingAdmin => SponsorError::NoPendingAdmin,
208        }
209    }
210}
