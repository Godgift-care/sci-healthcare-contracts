3/// Lifecycle state of a care provider in the registry.
4///
5/// A provider is `Pending` on self-registration and only becomes `Active`
6/// after an off-chain verification step performed by an attester or the
7/// admin. Vouchers can only be created against `Active` providers.
8///
9/// The lifecycle is a one-way street: `set_provider_status` only accepts
10/// the transitions `Pending → Active`, `Pending → Suspended`,
11/// `Active → Suspended` and `Suspended → Active` (anything else fails with
12/// `InvalidTransition`). A provider that leaves for good calls `retire`,
13/// which sets `Retired` — a terminal state nothing can move it out of.
14/// New statuses are added, never renumbered.
15#[contracttype]
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum ProviderStatus {
18    Pending = 0,
19    Active = 1,
20    Suspended = 2,
21    /// Withdrawn from the registry by its own owner via `retire`. Not
22    /// active, cannot receive vouchers, and no status change — by the admin
23    /// or anyone else — can leave it: re-registering is the way back.
24    Retired = 3,
25}
26
27impl ProviderStatus {
28    /// Whether the registry may move a provider from `from` to `to` with
29    /// `set_provider_status`.
30    ///
31    /// Only the four transitions that carry real information are allowed:
32    /// `Pending → Active` (verified), `Pending → Suspended` (refused before
33    /// it ever traded), `Active → Suspended` and `Suspended → Active`.
34    /// Everything else — including setting the same status twice — is a
35    /// no-op or backwards move that would emit a misleading event, and is
36    /// refused with `InvalidTransition`. Entering `Retired` is not one of
37    /// these: only the provider's own owner retires it, via `retire`.
38    pub fn is_transition_allowed(from: ProviderStatus, to: ProviderStatus) -> bool {
39        use ProviderStatus::{Active, Pending, Suspended};
40
41        matches!(
42            (from, to),
43            (Pending, Active) | (Pending, Suspended) | (Active, Suspended) | (Suspended, Active)
44        )
45    }
46}

48/// A registered care provider (clinic, pharmacy, diagnostic lab).
49///
50/// Deliberately contains no patient-identifying data. `owner` is the
51/// address that receives settlement for this provider's vouchers.
52#[contracttype]
53#[derive(Clone, Debug, PartialEq)]
54pub struct Provider {
55    pub owner: Address,
56    pub name: String,
57    /// ISO 3166-1 alpha-2 country code, e.g. "NG", "KE". Two ASCII
58    /// uppercase letters only.
59    pub country: String,
60    pub status: ProviderStatus,
61    pub registered_at: u64,
62    /// Reason code of the most recent status change, as documented in
63    /// `docs/protocol/trust.md`. Codes are added, never renumbered. 0 on
64    /// registration, before any status change has recorded a reason.
65    pub reason_code: u32,
66}

68/// A single billable service offered by a provider.
69///
70/// `code` is a coarse service *category* (e.g. outpatient consult, malaria
71/// RDT), never a diagnosis. Keeping this coarse is what keeps protected
72/// health information off the ledger.
73#[contracttype]
74#[derive(Clone, Debug, PartialEq)]
75pub struct ServiceItem {
76    pub code: u32,
77    pub label: String,
78    /// Price in the smallest unit of the settlement token.
79    pub price: i128,
80    pub active: bool,
81    /// Whether the clinic can deliver this service right now. A clinic that
82    /// runs out of a test kit pauses the service instead of delisting it, so
83    /// it keeps its catalogue entry. Defaults to `true` on first listing.
84    pub available: bool,
85    /// Price scheduled by `schedule_service_price`, or 0 if none. Replaces
86    /// `price` once the ledger reaches `effective_at`.
87    pub pending_price: i128,
88    /// When `pending_price` takes effect. 0 if nothing is scheduled.
89    pub effective_at: u64,
90}
91
92impl ServiceItem {
93    /// Applies a scheduled price whose time has come.
94    ///
95    /// Done on every read rather than by a keeper, so the new price takes
96    /// effect at exactly `effective_at` without anyone having to call in.
97    pub(crate) fn apply_due_price(&mut self, now: u64) {
98        if self.pending_price > 0 && now >= self.effective_at {
99            self.price = self.pending_price;
100            self.pending_price = 0;
101            self.effective_at = 0;
102        }
103    }
104}

129#[contracterror]
130#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
131#[repr(u32)]
132pub enum RegistryError {
133    AlreadyInitialized = 1,
134    NotInitialized = 2,
135    NotAuthorized = 3,
136    ProviderNotFound = 4,
137    ProviderExists = 5,
138    ProviderNotActive = 6,
139    ServiceNotFound = 7,
140    InvalidPrice = 8,
141    InvalidCountry = 9,
142    EmptyName = 10,
143    NoPendingAdmin = 11,
144    /// A scheduled price must give funders at least `MIN_PRICE_NOTICE_SECS`.
145    PriceNoticeTooShort = 12,
146    /// `upsert_service` may lower a listed price at once, but a rise has to
147    /// go through `schedule_service_price`.
148    PriceIncreaseNeedsNotice = 13,
149    /// The requested provider status change is not one of the allowed
150    /// transitions, or would leave `Retired`.
151    InvalidTransition = 14,
152    /// A scheduled price of `MAX_PRICE_JUMP` times the current price or more
153    /// needs the explicit `confirm_large_change` flag. Large jumps are almost
154    /// always a typo, and a typo here overcharges every funder that day.
155    LargePriceJumpNeedsConfirmation = 15,
156}

158/// Maps the shared helpers' failures onto this contract's error numbers, so
159/// callers keep seeing the same codes they did before the helpers were shared.
160impl From<sci_common::CommonError> for RegistryError {
161    fn from(error: sci_common::CommonError) -> Self {
162        match error {
163            sci_common::CommonError::NotAuthorized => RegistryError::NotAuthorized,
164            sci_common::CommonError::NoPendingAdmin => RegistryError::NoPendingAdmin,
165        }
166    }
167}
