3/// Lifecycle state of a care provider in the registry.
4///
5/// A provider is `Pending` on self-registration and only becomes `Active`
6/// after an off-chain verification step performed by an attester or the
7/// admin. Vouchers can only be created against `Active` providers.
8///
9/// The lifecycle is a one-way street: `set_provider_status` only accepts
10/// the transitions `Pending → Active`, `Pending → Suspended`,
11/// `Active → Suspended` and `Suspended → Active` (anything else fails with
12/// `InvalidTransition`). A provider that leaves for good calls `retire`,
13/// which sets `Retired` — a terminal state nothing can move it out of.
14/// New statuses are added, never renumbered.
15#[contracttype]
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum ProviderStatus {
18    Pending = 0,
19    Active = 1,
20    Suspended = 2,
21    /// Withdrawn from the registry by its own owner via `retire`. Not
22    /// active, cannot receive vouchers, and no status change — by the admin
23    /// or anyone else — can leave it: re-registering is the way back.
24    Retired = 3,
25}
26
27impl ProviderStatus {
28    /// Whether the registry may move a provider from `from` to `to` with
29    /// `set_provider_status`.
30    ///
31    /// Only the four transitions that carry real information are allowed:
32    /// `Pending → Active` (verified), `Pending → Suspended` (refused before
33    /// it ever traded), `Active → Suspended` and `Suspended → Active`.
34    /// Everything else — including setting the same status twice — is a
35    /// no-op or backwards move that would emit a misleading event, and is
36    /// refused with `InvalidTransition`. Entering `Retired` is not one of
37    /// these: only the provider's own owner retires it, via `retire`.
38    pub fn is_transition_allowed(from: ProviderStatus, to: ProviderStatus) -> bool {
39        use ProviderStatus::{Active, Pending, Suspended};
40
41        matches!(
42            (from, to),
43            (Pending, Active) | (Pending, Suspended) | (Active, Suspended) | (Suspended, Active)
44        )
45    }
46}

48/// A registered care provider (clinic, pharmacy, diagnostic lab).
49///
50/// Deliberately contains no patient-identifying data. `owner` is the
51/// address that receives settlement for this provider's vouchers.
52#[contracttype]
53#[derive(Clone, Debug, PartialEq)]
54pub struct Provider {
55    pub owner: Address,
56    pub name: String,
57    /// ISO 3166-1 alpha-2 country code, e.g. "NG", "KE". Two ASCII
58    /// uppercase letters only.
59    pub country: String,
60    pub status: ProviderStatus,
61    pub registered_at: u64,
62    /// Reason code of the most recent status change, as documented in
63    /// `docs/protocol/trust.md`. Codes are added, never renumbered. 0 on
64    /// registration, before any status change has recorded a reason.
65    pub reason_code: u32,
66    pub category: ProviderCategory,
67    /// Hash of the verification evidence bundle (e.g. licence, inspection report).
68    pub verification_evidence: Option<BytesN<32>>,
69}

82/// A single billable service offered by a provider.
83///
84/// `code` is a coarse service *category* (e.g. outpatient consult, malaria
85/// RDT), never a diagnosis. Keeping this coarse is what keeps protected
86/// health information off the ledger.
87#[contracttype]
88#[derive(Clone, Debug, PartialEq)]
89pub struct ServiceItem {
90    pub code: u32,
91    pub label: String,
92    /// Price in the smallest unit of the settlement token.
93    pub price: i128,
94    pub active: bool,
95    /// Whether the clinic can deliver this service right now. A clinic that
96    /// runs out of a test kit pauses the service instead of delisting it, so
97    /// it keeps its catalogue entry. Defaults to `true` on first listing.
98    pub available: bool,
99    /// Price scheduled by `schedule_service_price`, or 0 if none. Replaces
100    /// `price` once the ledger reaches `effective_at`.
101    pub pending_price: i128,
102    /// When `pending_price` takes effect. 0 if nothing is scheduled.
103    pub effective_at: u64,
104}
105
106impl ServiceItem {
107    /// Applies a scheduled price whose time has come.
108    ///
109    /// Done on every read rather than by a keeper, so the new price takes
110    /// effect at exactly `effective_at` without anyone having to call in.
111    pub(crate) fn apply_due_price(&mut self, now: u64) {
112        if self.pending_price > 0 && now >= self.effective_at {
113            self.price = self.pending_price;
114            self.pending_price = 0;
115            self.effective_at = 0;
116        }
117    }
118}

149#[contracterror]
150#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
151#[repr(u32)]
152pub enum RegistryError {
153    AlreadyInitialized = 1,
154    NotInitialized = 2,
155    NotAuthorized = 3,
156    ProviderNotFound = 4,
157    ProviderExists = 5,
158    ProviderNotActive = 6,
159    ServiceNotFound = 7,
160    InvalidPrice = 8,
161    InvalidCountry = 9,
162    EmptyName = 10,
163    NoPendingAdmin = 11,
164    /// A scheduled price must give funders at least `MIN_PRICE_NOTICE_SECS`.
165    PriceNoticeTooShort = 12,
166    /// `upsert_service` may lower a listed price at once, but a rise has to
167    /// go through `schedule_service_price`.
168    PriceIncreaseNeedsNotice = 13,
169    /// The requested provider status change is not one of the allowed
170    /// transitions, or would leave `Retired`.
171    InvalidTransition = 14,
172    /// A scheduled price of `MAX_PRICE_JUMP` times the current price or more
173    /// needs the explicit `confirm_large_change` flag. Large jumps are almost
174    /// always a typo, and a typo here overcharges every funder that day.
175    LargePriceJumpNeedsConfirmation = 15,
176    /// `set_provider_status` was called with `reason_code` 0. Code 0 means
177    /// "no status change yet" and is only ever written by registration, so a
178    /// real change — above all reinstating a suspended clinic — must say why.
179    MissingReasonCode = 16,
180}

182/// Maps the shared helpers' failures onto this contract's error numbers, so
183/// callers keep seeing the same codes they did before the helpers were shared.
184impl From<sci_common::CommonError> for RegistryError {
185    fn from(error: sci_common::CommonError) -> Self {
186        match error {
187            sci_common::CommonError::NotAuthorized => RegistryError::NotAuthorized,
188            sci_common::CommonError::NoPendingAdmin => RegistryError::NoPendingAdmin,
189        }
190    }
191}

209    /// Every status, in the row and column order of `ALLOWED`.
210    const ALL_STATUSES: [ProviderStatus; 4] = [
211        ProviderStatus::Pending,
212        ProviderStatus::Active,
213        ProviderStatus::Suspended,
214        ProviderStatus::Retired,
215    ];
216
217    /// The complete transition table, written out by hand: one row per `from`
218    /// status and one column per `to` status, in `ALL_STATUSES` order.
219    ///
220    /// This is deliberately not derived from `is_transition_allowed`: the
221    /// point is to state what the protocol allows independently, so a change
222    /// to the function has to be matched here or the test fails.
223    const ALLOWED: [[bool; 4]; 4] = [
224        // to: Pending, Active, Suspended, Retired
225        /* Pending   */
226        [false, true, true, false],
227        /* Active    */ [false, false, true, false],
228        /* Suspended */ [false, true, false, false],
229        /* Retired   */ [false, false, false, false],
230    ];
231
232    /// The row and column a status occupies in `ALLOWED`.
233    ///
234    /// The `match` is exhaustive on purpose: adding a `ProviderStatus`
235    /// variant makes this fail to compile, which means a status cannot be
236    /// added without extending the table above and the test below.
237    fn status_index(status: ProviderStatus) -> usize {
238        match status {
239            ProviderStatus::Pending => 0,
240            ProviderStatus::Active => 1,
241            ProviderStatus::Suspended => 2,
242            ProviderStatus::Retired => 3,
243        }
244    }
245
246    #[test]
247    fn provider_status_transition_table_is_exhaustive() {
248        for from in ALL_STATUSES {
249            for to in ALL_STATUSES {
250                let expected = ALLOWED[status_index(from)][status_index(to)];
251                assert_eq!(
252                    ProviderStatus::is_transition_allowed(from, to),
253                    expected,
254                    "is_transition_allowed({from:?}, {to:?}) disagrees with the table in \
255                     docs/developers/contracts.md. A status pair that is not written down \
256                     here is a pair nobody has agreed on",
257                );
258            }
259        }
260    }

277    fn service_item_due_price() {
278        let env = soroban_sdk::Env::default();
279        let mut item = ServiceItem {
280            code: 1,
281            label: String::from_str(&env, "Consult"),
282            price: 100,
283            active: true,
284            available: true,
285            pending_price: 200,
286            effective_at: 1000,
287        };
288        // Before effective_at: no change
289        item.apply_due_price(999);
290        assert_eq!(item.price, 100);
291        assert_eq!(item.pending_price, 200);
292
293        // At effective_at: price updated
294        item.apply_due_price(1000);
295        assert_eq!(item.price, 200);
296        assert_eq!(item.pending_price, 0);
297
298        // When pending_price is 0: no change
299        item.apply_due_price(2000);
300        assert_eq!(item.price, 200);
301    }
