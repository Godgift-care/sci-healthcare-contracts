8//! # SCI Healthcare — Provider Registry
9//!
10//! Answers one question for the rest of the protocol: *who is allowed to do
11//! what*. It holds verified care providers, the coarse service categories
12//! each one is approved to bill for, and the set of attesters permitted to
13//! confirm that care was delivered.
14//!
15//! No patient-identifying data is stored here, or anywhere else in this
16//! protocol. Services are coarse categories, never diagnoses.

74    /// Self-registers a provider in `Pending` state.
75    ///
76    /// Registration is permissionless; activation is not. A provider cannot
77    /// receive vouchers until an admin moves it to `Active`.
78    pub fn register_provider(
79        env: Env,
80        owner: Address,
81        name: String,
82        country: String,
83        category: ProviderCategory,
84    ) -> Result<(), RegistryError> {
85        owner.require_auth();
86
87        if name.is_empty() {
88            return Err(RegistryError::EmptyName);
89        }
90        // ISO 3166-1 alpha-2 is exactly two ASCII uppercase letters — never
91        // two arbitrary characters. A lowercase "ng", a digit, two spaces or
92        // a multi-byte character that happens to occupy two bytes would
93        // split one country into two entries in the app and the indexer, so
94        // it never enters the registry. The full ISO list is an off-chain
95        // concern; the shape is not.
96        let country_bytes = country.to_bytes();
97        if country_bytes.len() != 2 || !country_bytes.iter().all(|byte| byte.is_ascii_uppercase()) {
98            return Err(RegistryError::InvalidCountry);
99        }
100        if has_provider(&env, &owner) {
101            return Err(RegistryError::ProviderExists);
102        }
103
104        let registered_at = env.ledger().timestamp();
105        let provider = Provider {
106            owner: owner.clone(),
107            name: name.clone(),
108            country: country.clone(),
109            status: ProviderStatus::Pending,
110            registered_at,
111            reason_code: 0,
112            category,
113            verification_evidence: None,
114        };
115        write_provider(&env, &owner, &provider);
116        extend_instance(&env);
117
118        // Append to the registration index so any app can enumerate clinics
119        // without an indexer, in registration order.
120        let count = storage::read_provider_count(&env);
121        storage::write_provider_at(&env, count, &owner);
122        storage::write_provider_count(&env, &(count + 1));
123        storage::increment_status_count(&env, ProviderStatus::Pending);
124
125        // `CountryCount` for this country is left unset; `read_country_count`
126        // returns 0 when absent, so `providers_in` works even if the provider
127        // never becomes active. The entry is created when the provider is
128        // activated in `set_provider_status`.
129
130        ProviderRegistered {
131            provider: owner,
132            name,
133            country,
134            registered_at,
135            category,
136        }
137        .publish(&env);
138        Ok(())
139    }

162    /// Moves a provider between lifecycle states. Admin only.
163    ///
164    /// Only the four informative transitions are accepted — `Pending →
165    /// Active`, `Pending → Suspended`, `Active → Suspended` and
166    /// `Suspended → Active`. Setting the same status again, or moving a
167    /// provider backwards to `Pending`, is refused with `InvalidTransition`
168    /// instead of emitting an event that misleads the indexer and anyone
169    /// auditing the registry's history. Entering `Retired` is likewise not
170    /// the admin's call: the provider's own owner leaves with `retire`.
171    ///
172    /// `reason_code` is the documented reason for the change (see
173    /// `docs/protocol/trust.md`); it is stored on the provider and emitted
174    /// with the status, so a suspended clinic knows why and a funder can
175    /// tell a fraud suspension from a paperwork one. Code `0` is reserved for
176    /// "no status change yet" (written by registration) and is refused with
177    /// `MissingReasonCode`, so every change — reinstating a suspended clinic
178    /// included — records a real reason. Only the stored admin, signing, can
179    /// call this; the check runs before anything else is read.
180    ///
181    /// When a provider becomes `Active`, they are added to the per-country
182    /// index so `providers_in` can find them. When they become non-`Active`
183    /// (e.g. `Suspended`), they are removed from the index using swap-and-pop
184    /// so the index stays consistent and reads remain O(1) amortized.
185    pub fn set_provider_status(
186        env: Env,
187        admin: Address,
188        provider_addr: Address,
189        status: ProviderStatus,
190        reason_code: u32,
191    ) -> Result<(), RegistryError> {
192        Self::require_admin(&env, &admin)?;
193        if reason_code == 0 {
194            return Err(RegistryError::MissingReasonCode);
195        }
196
197        let mut provider = read_provider(&env, &provider_addr)?;
198        if !ProviderStatus::is_transition_allowed(provider.status, status) {
199            return Err(RegistryError::InvalidTransition);
200        }
201        let prev_status = provider.status;
202        provider.status = status;
203        provider.reason_code = reason_code;
204        write_provider(&env, &provider_addr, &provider);
205        extend_instance(&env);
206
207        // Update the per-country index when crossing the Active boundary.
208        if prev_status != status {
209            storage::decrement_status_count(&env, prev_status);
210            storage::increment_status_count(&env, status);
211            let country = &provider.country;
212            match (prev_status, status) {
213                // Provider just became active: add to country index.
214                (_, ProviderStatus::Active) => {
215                    let count = storage::read_country_count(&env, country);
216                    storage::write_country_at(&env, country, count, &provider_addr);
217                    storage::write_country_count(&env, country, &(count + 1));
218                }
219                // Provider just became non-active: remove from country index
220                // using swap-and-pop.
221                (ProviderStatus::Active, _) => {
222                    // Find the provider's index in the country list.
223                    let total = storage::read_country_count(&env, country);
224                    let mut idx = None;
225                    for i in 0..total {
226                        if let Some(addr) = storage::read_country_at(&env, country, i) {
227                            if addr == provider_addr {
228                                idx = Some(i);
229                                break;
230                            }
231                        }
232                    }
233                    if let Some(i) = idx {
234                        // Swap with the last element and pop.
235                        let last_idx = total.saturating_sub(1);
236                        if i != last_idx {
237                            if let Some(last_addr) =
238                                storage::read_country_at(&env, country, last_idx)
239                            {
240                                storage::write_country_at(&env, country, i, &last_addr);
241                            }
242                        }
243                        storage::write_country_count(&env, country, &(total - 1));
244                    }
245                }
246                // Reactivation (Pending -> Active or Suspended -> Active) is
247                // handled above; no double-add because we only act on the
248                // transition from non-Active to Active.
249                _ => {}
250            }
251        }
252
253        ProviderStatusChanged {
254            provider: provider_addr,
255            status,
256            reason_code,
257        }
258        .publish(&env);
259        Ok(())
260    }

287    /// Lets a provider withdraw from the registry. Its own owner only.
288    ///
289    /// Clinics close, and a closed clinic that stays listed keeps drawing
290    /// funders' money into escrow it will never claim. `retire` sets the
291    /// terminal `Retired` status: the clinic immediately stops being an
292    /// active provider, so `voucher` refuses to fund it any new vouchers
293    /// and an unclaimed one refunds without waiting for expiry, while a
294    /// voucher claimed before the clinic retired still attests and
295    /// settles as care delivered.
296    ///
297    /// Nothing can move a retired clinic back in, not even the admin: it
298    /// must register again. Retiring an already-retired clinic is refused
299    /// with `InvalidTransition`, so the lifecycle history stays free of
300    /// no-op events. `reason_code` follows the same documented codes as
301    /// `set_provider_status` (see `docs/protocol/trust.md`).
302    pub fn retire(env: Env, owner: Address, reason_code: u32) -> Result<(), RegistryError> {
303        owner.require_auth();
304
305        let mut provider = read_provider(&env, &owner)?;
306        if provider.status == ProviderStatus::Retired {
307            return Err(RegistryError::InvalidTransition);
308        }
309        let prev_status = provider.status;
310        provider.status = ProviderStatus::Retired;
311        provider.reason_code = reason_code;
312        write_provider(&env, &owner, &provider);
313        extend_instance(&env);
314
315        storage::decrement_status_count(&env, prev_status);
316        storage::increment_status_count(&env, ProviderStatus::Retired);
317
318        ProviderStatusChanged {
319            provider: owner,
320            status: ProviderStatus::Retired,
321            reason_code,
322        }
323        .publish(&env);
324        Ok(())
325    }

327    /// Creates or updates a service the provider bills for.
328    ///
329    /// Only an `Active` provider may maintain a catalog, so a suspended
330    /// clinic cannot quietly re-price itself back into circulation.
331    ///
332    /// A new service is listed at `price` straight away, and so is a lower
333    /// price for an existing one. Raising an existing price is rejected with
334    /// `PriceIncreaseNeedsNotice`: use `schedule_service_price`, so a funder
335    /// never sees one price and finds another minutes later. A price already
336    /// scheduled is left in place.
337    pub fn upsert_service(
338        env: Env,
339        provider_addr: Address,
340        code: u32,
341        label: String,
342        price: i128,
343    ) -> Result<(), RegistryError> {
344        provider_addr.require_auth();
345
346        if price <= 0 {
347            return Err(RegistryError::InvalidPrice);
348        }
349        if label.is_empty() {
350            return Err(RegistryError::EmptyName);
351        }
352
353        let provider = read_provider(&env, &provider_addr)?;
354        if provider.status != ProviderStatus::Active {
355            return Err(RegistryError::ProviderNotActive);
356        }
357
358        let item = match read_service(&env, &provider_addr, code) {
359            Ok(existing) => {
360                if price > existing.price {
361                    return Err(RegistryError::PriceIncreaseNeedsNotice);
362                }
363                // Re-pricing or relabelling a paused service must not quietly
364                // put it back on sale; only `set_service_available` does that.
365                ServiceItem {
366                    label: label.clone(),
367                    price,
368                    active: true,
369                    ..existing
370                }
371            }
372            Err(_) => ServiceItem {
373                code,
374                label: label.clone(),
375                price,
376                active: true,
377                available: true,
378                pending_price: 0,
379                effective_at: 0,
380            },
381        };
382        write_service(&env, &provider_addr, code, &item);
383        extend_instance(&env);
384
385        ServiceUpserted {
386            provider: provider_addr,
387            code,
388            label,
389            price,
390        }
391        .publish(&env);
392        Ok(())
393    }

395    /// Schedules a new price for a listed service, effective at `effective_at`.
396    ///
397    /// The time must be at least `MIN_PRICE_NOTICE_SECS` away, so apps can
398    /// warn funders before it lands. Until then `get_service_price` keeps
399    /// returning the current price; from then on it returns this one.
400    /// Scheduling again replaces the earlier schedule, and scheduling the
401    /// current price is how a clinic calls one off.
402    ///
403    /// A price of `MAX_PRICE_JUMP` times the current price or more is rejected
404    /// with `LargePriceJumpNeedsConfirmation` unless `confirm_large_change`
405    /// is `true`. A 5x increase is almost always a mistyped zero, and every
406    /// funder that day would pay it, so the clinic has to say so explicitly.
407    pub fn schedule_service_price(
408        env: Env,
409        provider_addr: Address,
410        code: u32,
411        price: i128,
412        effective_at: u64,
413        confirm_large_change: bool,
414    ) -> Result<(), RegistryError> {
415        provider_addr.require_auth();
416
417        if price <= 0 {
418            return Err(RegistryError::InvalidPrice);
419        }
420        let provider = read_provider(&env, &provider_addr)?;
421        if provider.status != ProviderStatus::Active {
422            return Err(RegistryError::ProviderNotActive);
423        }
424        let mut item = read_service(&env, &provider_addr, code)?;
425
426        // A jump of MAX_PRICE_JUMP times or more is treated as a typo until
427        // the clinic acknowledges it. `checked_mul` means an overflow is also
428        // a large jump rather than a panic.
429        let threshold = item.price.checked_mul(MAX_PRICE_JUMP as i128);
430        match threshold {
431            Some(limit) if price < limit => {}
432            _ if confirm_large_change => {}
433            _ => return Err(RegistryError::LargePriceJumpNeedsConfirmation),
434        }
435
436        let earliest = env
437            .ledger()
438            .timestamp()
439            .saturating_add(MIN_PRICE_NOTICE_SECS);
440        if effective_at < earliest {
441            return Err(RegistryError::PriceNoticeTooShort);
442        }
443
444        item.pending_price = price;
445        item.effective_at = effective_at;
446        write_service(&env, &provider_addr, code, &item);
447        extend_instance(&env);
448
449        ServicePriceScheduled {
450            provider: provider_addr,
451            code,
452            price,
453            effective_at,
454        }
455        .publish(&env);
456        Ok(())
457    }

551    /// Revokes attester rights. Admin only.
552    pub fn remove_attester(
553        env: Env,
554        admin: Address,
555        attester: Address,
556    ) -> Result<(), RegistryError> {
557        Self::require_admin(&env, &admin)?;
558        if read_attester(&env, &attester) {
559            write_attester(&env, &attester, false);
560            let total = storage::read_attester_count(&env);
561            let mut idx = None;
562            for i in 0..total {
563                if let Some(addr) = storage::read_attester_at(&env, i) {
564                    if addr == attester {
565                        idx = Some(i);
566                        break;
567                    }
568                }
569            }
570            if let Some(i) = idx {
571                let last_idx = total.saturating_sub(1);
572                if i != last_idx {
573                    if let Some(last_addr) = storage::read_attester_at(&env, last_idx) {
574                        storage::write_attester_at(&env, i, &last_addr);
575                    }
576                }
577                storage::remove_attester_at(&env, last_idx);
578                storage::write_attester_count(&env, &(total - 1));
579            }
580        }
581        extend_instance(&env);
582        AttesterRemoved { attester }.publish(&env);
583        Ok(())
584    }

679    /// Batched so a client can price a whole catalog in one RPC round trip
680    /// instead of one call per service.
681    pub fn get_services(env: Env, provider_addr: Address, codes: Vec<u32>) -> Vec<ServiceItem> {
682        let mut out = Vec::new(&env);
683        for code in codes.iter() {
684            if let Ok(item) = read_service(&env, &provider_addr, code) {
685                if item.active {
686                    out.push_back(item);
687                }
688            }
689        }
690        out
691    }

693    /// Price for a provider's service, or 0 if absent or inactive.
694    ///
695    /// A scheduled price is returned from its `effective_at` onwards.
696    ///
697    /// A paused service still reports its price, so `voucher` can tell
698    /// "not offered" apart from "temporarily unavailable".
699    ///
700    /// Returns a primitive rather than a struct so the voucher contract can
701    /// validate pricing over a cross-contract interface without having to
702    /// redeclare this contract's types.
703    pub fn get_service_price(env: Env, provider_addr: Address, code: u32) -> i128 {
704        match read_service(&env, &provider_addr, code) {
705            Ok(item) if item.active => item.price,
706            _ => 0,
707        }
708    }

710    /// The scheduled price and when it takes effect, or `(0, 0)` if no
711    /// change is pending. Lets apps warn funders about an upcoming price.
712    pub fn get_upcoming_price(
713        env: Env,
714        provider_addr: Address,
715        code: u32,
716    ) -> Result<(i128, u64), RegistryError> {
717        let item = read_service(&env, &provider_addr, code)?;
718        Ok((item.pending_price, item.effective_at))
719    }
