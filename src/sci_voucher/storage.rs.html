10/// How a voucher is laid out in storage.
11///
12/// `Voucher` is a named struct, which the host encodes as an XDR map: every
13/// field is stored with its name as a symbol key, so twelve field names ride
14/// along in every voucher entry and are paid for in rent for its whole life.
15/// This tuple struct is encoded as an XDR vec — values only, in order — and
16/// leaves out the id, which is already in the entry's key. The public API
17/// still takes and returns `Voucher`; only the bytes at rest change.
18///
19/// Field order is part of the storage format: never reorder, only append.
20///
21/// Not in the contract spec: it is `pub(crate)` and never crosses the
22/// contract boundary, which is what soroban-sdk 29 keys the spec off.
23#[contracttype]
24#[derive(Clone, Debug, PartialEq)]
25pub(crate) struct StoredVoucher(
26    Address,         // funder
27    BytesN<32>,      // beneficiary_ref
28    Address,         // provider
29    u32,             // service_code
30    i128,            // amount
31    VoucherStatus,   // status
32    u64,             // created_at
33    u64,             // expires_at
34    u64,             // claimed_at
35    u64,             // attested_at
36    u64,             // dispute_deadline
37    Option<Address>, // attester
38);
39
40impl StoredVoucher {
41    pub(crate) fn from_voucher(v: &Voucher) -> Self {
42        StoredVoucher(
43            v.funder.clone(),
44            v.beneficiary_ref.clone(),
45            v.provider.clone(),
46            v.service_code,
47            v.amount,
48            v.status,
49            v.created_at,
50            v.expires_at,
51            v.claimed_at,
52            v.attested_at,
53            v.dispute_deadline,
54            v.attester.clone(),
55        )
56    }
57
58    pub(crate) fn into_voucher(self, id: u64) -> Voucher {
59        Voucher {
60            id,
61            funder: self.0,
62            beneficiary_ref: self.1,
63            provider: self.2,
64            service_code: self.3,
65            amount: self.4,
66            status: self.5,
67            created_at: self.6,
68            expires_at: self.7,
69            claimed_at: self.8,
70            attested_at: self.9,
71            dispute_deadline: self.10,
72            attester: self.11,
73        }
74    }
75}

107pub(crate) fn read_voucher(env: &Env, id: u64) -> Result<Voucher, VoucherError> {
108    let key = DataKey::Voucher(id);
109    let stored: StoredVoucher = env
110        .storage()
111        .persistent()
112        .get(&key)
113        .ok_or(VoucherError::VoucherNotFound)?;
114    common::extend_persistent(env, &key);
115    Ok(stored.into_voucher(id))
116}

118/// Writes a voucher and keeps the contract instance alive with it.
119///
120/// Every state transition goes through here, so a contract that is only
121/// ever claimed, attested and settled (never funded) for a month still has
122/// its config and id counter extended.
123pub(crate) fn write_voucher(env: &Env, voucher: &Voucher) {
124    let key = DataKey::Voucher(voucher.id);
125    env.storage()
126        .persistent()
127        .set(&key, &StoredVoucher::from_voucher(voucher));
128    common::extend_persistent(env, &key);
129    extend_instance(env);
130}

145/// Adds newly escrowed funds to the running total.
146pub(crate) fn increase_open_escrow(env: &Env, amount: i128) -> Result<(), VoucherError> {
147    let total = read_open_escrow(env)
148        .checked_add(amount)
149        .ok_or(VoucherError::MathOverflow)?;
150    write_open_escrow(env, total);
151    Ok(())
152}
153
154/// Removes funds that have left escrow, either to the provider or the funder.
155pub(crate) fn decrease_open_escrow(env: &Env, amount: i128) -> Result<(), VoucherError> {
156    let total = read_open_escrow(env)
157        .checked_sub(amount)
158        .ok_or(VoucherError::MathOverflow)?;
159    write_open_escrow(env, total);
160    Ok(())
161}

190/// A voucher's options; all off when it was funded without any.
191///
192/// The entry lives and dies with the voucher, so it is extended whenever it
193/// is read, exactly like the voucher itself.
194pub(crate) fn read_options(env: &Env, id: u64) -> VoucherOptions {
195    let key = DataKey::Options(id);
196    match env.storage().persistent().get(&key) {
197        Some(options) => {
198            common::extend_persistent(env, &key);
199            options
200        }
201        None => VoucherOptions::default(),
202    }
203}
204
205/// Stores a voucher's options — only when one is set, so the common voucher
206/// gets no second entry.
207pub(crate) fn write_options(env: &Env, id: u64, options: &VoucherOptions) {
208    if *options == VoucherOptions::default() {
209        return;
210    }
211    let key = DataKey::Options(id);
212    env.storage().persistent().set(&key, options);
213    common::extend_persistent(env, &key);
214}

264    fn write_options_with_default_options_returns_early() {
265        let env = Env::default();
266        let contract_id = env.register(crate::VoucherEscrow, ());
267        env.as_contract(&contract_id, || {
268            let id = 1u64;
269            let options = VoucherOptions::default();
