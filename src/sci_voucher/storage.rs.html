10/// How a voucher is laid out in storage.
11///
12/// `Voucher` is a named struct, which the host encodes as an XDR map: every
13/// field is stored with its name as a symbol key, so thirteen field names ride
14/// along in every voucher entry and are paid for in rent for its whole life.
15/// This tuple struct is encoded as an XDR vec — values only, in order — and
16/// leaves out the id, which is already in the entry's key. The public API
17/// still takes and returns `Voucher`; only the bytes at rest change.
18///
19/// Field order is part of the storage format: never reorder, only append.
20///
21/// Not in the contract spec: it is `pub(crate)` and never crosses the
22/// contract boundary, which is what soroban-sdk 29 keys the spec off.
23#[contracttype]
24#[derive(Clone, Debug, PartialEq)]
25pub(crate) struct StoredVoucher(
26    Address,         // funder
27    BytesN<32>,      // beneficiary_ref
28    Address,         // provider
29    u32,             // service_code
30    i128,            // amount
31    VoucherStatus,   // status
32    u64,             // created_at
33    u64,             // expires_at
34    u64,             // claimed_at
35    u64,             // attested_at
36    u64,             // dispute_deadline
37    Option<Address>, // attester
38    u32,             // updated_ledger
39);
40
41/// The positional layout before `updated_ledger` was appended.
42///
43/// Kept only so a voucher written by an older deployment still decodes. New
44/// writes always use `StoredVoucher`; a read falls back to this and reports
45/// `updated_ledger = 0`, meaning the ledger is unknown.
46#[contracttype]
47#[derive(Clone)]
48pub(crate) struct StoredVoucherV1(
49    pub(crate) Address,
50    pub(crate) BytesN<32>,
51    pub(crate) Address,
52    pub(crate) u32,
53    pub(crate) i128,
54    pub(crate) VoucherStatus,
55    pub(crate) u64,
56    pub(crate) u64,
57    pub(crate) u64,
58    pub(crate) u64,
59    pub(crate) u64,
60    pub(crate) Option<Address>,
61);

153/// Reads a voucher without changing its TTL.
154///
155/// Views that may touch many vouchers use this: a read-only `statuses` call
156/// must not charge rent or bump entries it only looks at. `read_voucher`
157/// wraps it for the normal single-voucher path.
158pub(crate) fn read_voucher_without_extend(env: &Env, id: u64) -> Result<Voucher, VoucherError> {
159    let key = DataKey::Voucher(id);
160    let raw: Val = env
161        .storage()
162        .persistent()
163        .get(&key)
164        .ok_or(VoucherError::VoucherNotFound)?;
165
166    // Check the encoded length before converting: unpacking a 12-field vec
167    // into the 13-field `StoredVoucher` traps inside the host rather than
168    // returning a conversion error, and an old voucher must still decode.
169    let fields = Vec::<Val>::try_from_val(env, &raw).map_err(|_| VoucherError::VoucherNotFound)?;
170    if fields.len() as usize == 13 {
171        return StoredVoucher::try_from_val(env, &raw)
172            .map(|stored| stored.into_voucher(id))
173            .map_err(|_| VoucherError::VoucherNotFound);
174    }
175    if fields.len() as usize == 12 {
176        return StoredVoucherV1::try_from_val(env, &raw)
177            .map(|old| old.into_voucher(id))
178            .map_err(|_| VoucherError::VoucherNotFound);
179    }
180    Err(VoucherError::VoucherNotFound)
181}

183pub(crate) fn read_voucher(env: &Env, id: u64) -> Result<Voucher, VoucherError> {
184    let voucher = read_voucher_without_extend(env, id)?;
185    common::extend_persistent(env, &DataKey::Voucher(id));
186    Ok(voucher)
187}

189/// Writes a voucher and keeps the contract instance alive with it.
190///
191/// Every state transition goes through here, so a contract that is only
192/// ever claimed, attested and settled (never funded) for a month still has
193/// its config and id counter extended. The write also stamps the current
194/// ledger sequence onto the voucher, which is why callers pass it mutably.
195pub(crate) fn write_voucher(env: &Env, voucher: &mut Voucher) {
196    voucher.updated_ledger = env.ledger().sequence();
197    let key = DataKey::Voucher(voucher.id);
198    env.storage()
199        .persistent()
200        .set(&key, &StoredVoucher::from_voucher(voucher));
201    common::extend_persistent(env, &key);
202    extend_instance(env);
203}

218/// Adds newly escrowed funds to the running total.
219pub(crate) fn increase_open_escrow(env: &Env, amount: i128) -> Result<(), VoucherError> {
220    let total = read_open_escrow(env)
221        .checked_add(amount)
222        .ok_or(VoucherError::MathOverflow)?;
223    write_open_escrow(env, total);
224    Ok(())
225}

227/// Removes funds that have left escrow, either to the provider or the funder.
228pub(crate) fn decrease_open_escrow(env: &Env, amount: i128) -> Result<(), VoucherError> {
229    let total = read_open_escrow(env)
230        .checked_sub(amount)
231        .ok_or(VoucherError::MathOverflow)?;
232    write_open_escrow(env, total);
233    Ok(())
234}

263/// A voucher's options; all off when it was funded without any.
264///
265/// The entry lives and dies with the voucher, so it is extended whenever it
266/// is read, exactly like the voucher itself.
267pub(crate) fn read_options(env: &Env, id: u64) -> VoucherOptions {
268    let key = DataKey::Options(id);
269    match env.storage().persistent().get(&key) {
270        Some(options) => {
271            common::extend_persistent(env, &key);
272            options
273        }
274        None => VoucherOptions::default(),
275    }
276}
277
278/// Stores a voucher's options — only when one is set, so the common voucher
279/// gets no second entry.
280pub(crate) fn write_options(env: &Env, id: u64, options: &VoucherOptions) {
281    if *options == VoucherOptions::default() {
282        return;
283    }
284    let key = DataKey::Options(id);
285    env.storage().persistent().set(&key, options);
286    common::extend_persistent(env, &key);
287}
