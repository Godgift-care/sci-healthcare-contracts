3/// Lifecycle of a single care voucher.
4///
5/// ```text
6///                  claim            attest          settle
7///   Funded ──────────────> Claimed ────────> Attested ──────> Settled
8///     │                       │                  │
9///     │ refund                │ dispute          │ dispute
10///     │ (expired or           ▼                  ▼
11///     │  clinic inactive)  Disputed <────────────┘
12///     ▼                       │
13///  Refunded  <──── resolve ───┴──── resolve ────> Settled
14///     ▲
15///     └── refund (claimed, never attested, past expiry + claim grace)
16/// ```
17///
18/// `Settled` and `Refunded` are terminal.
19#[contracttype]
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum VoucherStatus {
22    Funded = 0,
23    Claimed = 1,
24    Attested = 2,
25    Settled = 3,
26    Disputed = 4,
27    Refunded = 5,
28}

30/// An escrowed commitment to pay one provider for one service.
31///
32/// The funder is whoever paid — the patient themselves, or a family member
33/// sending from abroad. `beneficiary_ref` is an opaque salted hash of an
34/// off-chain patient identifier; it carries no clinical meaning and cannot
35/// be reversed into an identity from on-chain data alone.
36#[contracttype]
37#[derive(Clone, Debug, PartialEq)]
38pub struct Voucher {
39    pub id: u64,
40    pub funder: Address,
41    pub beneficiary_ref: BytesN<32>,
42    pub provider: Address,
43    pub service_code: u32,
44    pub amount: i128,
45    pub status: VoucherStatus,
46    pub created_at: u64,
47    pub expires_at: u64,
48    pub claimed_at: u64,
49    pub attested_at: u64,
50    /// Earliest ledger time at which `settle` may run. Zero until attested.
51    pub dispute_deadline: u64,
52    /// Attester who confirmed delivery, recorded by `attest`.
53    ///
54    /// `None` until attestation. Settlement pays this address its share of
55    /// the protocol fee, which is why it is stored on the voucher rather than
56    /// looked up later: the attester of record is the one who did the work.
57    pub attester: Option<Address>,
58    /// Ledger sequence of the last write to this voucher: funding, claim,
59    /// attestation, dispute, settlement, refund, redirect or reduction.
60    ///
61    /// Indexers reconcile against ledger sequence rather than ledger time,
62    /// which is only approximate. Vouchers written before this field existed
63    /// decode with `0`, meaning the ledger is unknown.
64    pub updated_ledger: u32,
65}

67/// Optional terms a funder can attach to a voucher with
68/// `create_voucher_with_options`. Both default to off, and a voucher funded
69/// with `create_voucher` behaves exactly as before.
70#[contracttype]
71#[derive(Clone, Debug, Default, PartialEq)]
72pub struct VoucherOptions {
73    /// SHA-256 of a secret the funder gives the patient. When set, the clinic
74    /// can only claim with `claim_with_proof`, revealing the secret, which
75    /// shows it had the patient in front of it.
76    pub presence_commitment: Option<BytesN<32>>,
77    /// Seconds the funder has to dispute after attestation, in place of the
78    /// protocol-wide `Config.dispute_window`. Must lie within the dispute
79    /// window bounds.
80    pub dispute_window: Option<u64>,
81}

83/// Protocol-level settings, fixed at initialization.
84#[contracttype]
85#[derive(Clone, Debug, PartialEq)]
86pub struct Config {
87    pub admin: Address,
88    pub registry: Address,
89    pub receipt_book: Address,
90    /// Settlement token. USDC on Stellar in production.
91    pub token: Address,
92    /// Seconds a funder has to dispute after attestation.
93    pub dispute_window: u64,
94    /// Protocol fee in basis points. 100 = 1%.
95    pub fee_bps: u32,
96    /// Share of the voucher amount paid to the attester, in basis points.
97    ///
98    /// Carved out of `fee_bps`, so it can never exceed the protocol fee:
99    /// the clinic's net is unchanged, the attester receives
100    /// `amount * attester_share_bps / 10_000`, and the fee account receives
101    /// the rest. Zero means attesters are not paid at settlement, which is
102    /// exactly the pre-attester-share behaviour.
103    pub attester_share_bps: u32,
104    pub fee_account: Address,
105}

107#[contracttype]
108#[derive(Clone)]
109pub enum DataKey {
110    Config,
111    NextId,
112    Voucher(u64),
113    /// Address nominated by `propose_admin`, awaiting `accept_admin`.
114    PendingAdmin,
115    /// Most attestations one attester may make per UTC day. Unset means
116    /// `DEFAULT_MAX_ATTESTATIONS_PER_DAY`.
117    AttestationCap,
118    /// Attestations made by an attester on a UTC day (`timestamp / 86400`).
119    /// Temporary storage: it only has to outlive the day it counts.
120    Attestations(Address, u64),
121    /// A voucher's `VoucherOptions`. Written only for vouchers that set one,
122    /// so a plain voucher pays for no extra storage.
123    Options(u64),
124    /// Shortest and longest per-voucher dispute window a funder may choose.
125    /// Unset means `DEFAULT_DISPUTE_WINDOW_BOUNDS`.
126    DisputeWindowBounds,
127    /// Total amount held in escrow by vouchers that are not yet settled or
128    /// refunded. Kept in instance storage and updated on every movement of
129    /// funds, so `escrow_summary` can report it without scanning vouchers.
130    OpenEscrow,
131    /// What a clinic currently has in escrow: see `Exposure`.
132    Exposure(Address),
133    /// Most value one clinic may hold in escrow at once. Unset or 0 means
134    /// unlimited.
135    EscrowCap,
136    /// Value above which a voucher must carry a presence commitment. Unset
137    /// or 0 means no protocol-level requirement.
138    PresenceThreshold,
139}

141/// A clinic's open vouchers: every voucher funded to it that has not yet
142/// been settled or refunded, counted and summed.
143///
144/// Raised by `create_voucher` and lowered when a voucher settles or refunds,
145/// so it backs both the clinic's declared capacity and the protocol's
146/// per-clinic escrow cap.
147#[contracttype]
148#[derive(Clone, Debug, Default, PartialEq)]
149pub struct Exposure {
150    pub vouchers: u32,
151    pub value: i128,
152}

154#[contracterror]
155#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
156#[repr(u32)]
157pub enum VoucherError {
158    AlreadyInitialized = 1,
159    NotInitialized = 2,
160    NotAuthorized = 3,
161    VoucherNotFound = 4,
162    ProviderNotActive = 5,
163    ServiceNotOffered = 6,
164    AmountBelowPrice = 7,
165    InvalidAmount = 8,
166    InvalidExpiry = 9,
167    InvalidStatus = 10,
168    VoucherExpired = 11,
169    NotYetExpired = 12,
170    DisputeWindowOpen = 13,
171    DisputeWindowClosed = 14,
172    InvalidFee = 15,
173    MathOverflow = 16,
174    NoPendingAdmin = 17,
175    /// The clinic has paused this service; it cannot be funded until it
176    /// is made available again.
177    ServiceUnavailable = 18,
178    /// This attester has used up today's attestations; try again tomorrow
179    /// (UTC) or ask the admin to raise the cap.
180    AttestationCapReached = 19,
181    /// The attestation cap must be at least 1.
182    InvalidAttestationCap = 20,
183    /// `attester_share_bps` was greater than `fee_bps`. The attester's share
184    /// is carved out of the protocol fee, so it cannot exceed it.
185    InvalidAttesterShare = 21,
186    /// The new amount is not lower than the amount already escrowed, so
187    /// there is nothing to give back.
188    AmountNotReduced = 22,
189    /// The contract holds less of the settlement token than this voucher
190    /// escrows. Happens only if the token issuer clawed escrowed funds back.
191    EscrowShortfall = 23,
192    /// The token refused the payment, typically because the issuer froze
193    /// (deauthorized) the escrow or the recipient. Nothing moved; the voucher
194    /// is unchanged and the call can be retried once the freeze is lifted.
195    TokenTransferFailed = 24,
196    /// The clinic already has as many vouchers outstanding as the capacity it
197    /// declared in the registry.
198    ProviderAtCapacity = 25,
199    /// Funding this voucher would take the clinic's escrow above the
200    /// per-clinic cap.
201    EscrowCapReached = 26,
202    /// The per-clinic escrow cap cannot be negative.
203    InvalidEscrowCap = 27,
204    /// This voucher was funded with a presence commitment, so the clinic must
205    /// claim it with `claim_with_proof`.
206    PresenceProofRequired = 28,
207    /// The secret does not hash to the voucher's presence commitment.
208    InvalidPresenceProof = 29,
209    /// `claim_with_proof` on a voucher funded without a presence commitment.
210    NoPresenceCommitment = 30,
211    /// The chosen dispute window is outside the allowed bounds, or the bounds
212    /// themselves are invalid.
213    InvalidDisputeWindow = 31,
214    /// This voucher is above the admin-set presence threshold, so it must be
215    /// funded with a `presence_commitment`.
216    PresenceCommitmentRequired = 32,
217    /// The presence threshold cannot be negative.
218    InvalidPresenceThreshold = 33,
219    /// The settlement token moved a different amount than requested: the
220    /// escrow received less (or more) than a funder paid in, or a payout
221    /// debited the escrow or credited the recipient by something other than
222    /// the amount sent. Happens only with a token that charges transfer fees
223    /// or runs hooks; USDC on Stellar never does. The whole call reverts.
224    TokenAmountMismatch = 34,
225    /// The registry price changed between the quote the funder saw and this
226    /// call. Nothing was escrowed; re-quote before signing.
227    PriceChanged = 42,
228}

230/// Maps the shared helpers' failures onto this contract's error numbers, so
231/// callers keep seeing the same codes they did before the helpers were shared.
232impl From<sci_common::CommonError> for VoucherError {
233    fn from(error: sci_common::CommonError) -> Self {
234        match error {
235            sci_common::CommonError::NotAuthorized => VoucherError::NotAuthorized,
236            sci_common::CommonError::NoPendingAdmin => VoucherError::NoPendingAdmin,
237        }
238    }
239}
