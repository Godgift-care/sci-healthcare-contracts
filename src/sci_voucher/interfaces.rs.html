1// The generated client for `ReceiptInterface::mint` carries the same argument
2// list as the contract, which is longer than clippy's default budget.
3#![allow(clippy::too_many_arguments)]
4
5use soroban_sdk::{contractclient, Address, BytesN, Env};
6
7/// The slice of `sci-registry` this contract depends on.
8///
9/// Declared locally, over primitives only, so the two contracts share no
10/// types and can be upgraded independently.
11#[allow(dead_code)]
12#[contractclient(name = "RegistryClient")]
13pub trait RegistryInterface {
14    fn is_active_provider(env: Env, provider_addr: Address) -> bool;
15    fn is_attester(env: Env, addr: Address) -> bool;
16    fn get_service_price(env: Env, provider_addr: Address, code: u32) -> i128;
17    fn is_service_available(env: Env, provider_addr: Address, code: u32) -> bool;
18    fn get_capacity(env: Env, provider_addr: Address) -> u32;
19}
20
21/// The slice of `sci-receipt` this contract depends on.
22#[allow(dead_code, clippy::too_many_arguments)]
23#[contractclient(name = "ReceiptClient")]
24pub trait ReceiptInterface {
25    fn mint(
26        env: Env,
27        minter: Address,
28        voucher_id: u64,
29        beneficiary_ref: BytesN<32>,
30        provider: Address,
31        service_code: u32,
32        amount: i128,
33        fee: i128,
34        net: i128,
35    );
36}
